
# For more information about using CMake with Android Studio, read the
# documentation: https://d.android.com/studio/projects/add-native-code.html.
# For more examples on how to use CMake, see https://github.com/android/ndk-samples.

# Sets the minimum CMake version required for this project.
cmake_minimum_required(VERSION 3.22.1)

# Declares the project name. The project name can be accessed via ${ PROJECT_NAME},
# Since this is the top level CMakeLists.txt, the project name is also accessible
# with ${CMAKE_PROJECT_NAME} (both CMake variables are in-sync within the top level
# build script scope).
project("voiceinput")

set(SKBUILD ON)

# TODO: Double check this works on devices and doesn't crash if fp16 is unavailable
if(ANDROID_ABI STREQUAL "armeabi-v7a")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -mfpu=neon-fp16 -mfloat-abi=softfp")
endif()

# Custom build: let ggml use the ARMv8.2 dot-product and fp16 instructions, which every
# flagship phone since 2018 has (including the S25 Ultra). This makes quantized speech
# models much faster on the CPU. (Baseline ARMv8 is used otherwise.)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
    set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "" FORCE)
endif()

# Custom build: GPU backend for the speech engine on Android = OpenCL (generic ggml kernels). (Vulkan was tried first: on the Adreno 830 it produced empty
# transcripts and was slower than the CPU, so it is not built.)
#
# - CL headers come from the pinned Khronos OpenCL-Headers submodule.
# - ggml-opencl links to libvoice_opencl_shim.so (opencl_shim.c), which loads the phone's own
#   OpenCL driver at runtime and forwards calls to it. If the driver isn't available, the shim
#   reports no platforms and voice input runs on the CPU; the engine always loads.
option(VOICE_ENABLE_OPENCL "Build the OpenCL GPU backend for voice input" ON)
if(ANDROID AND ANDROID_ABI STREQUAL "arm64-v8a" AND VOICE_ENABLE_OPENCL)
    # opencl_shim.c is generated by tools/gen_opencl_shim.py (see that file)
    add_library(voice_opencl_shim SHARED opencl_shim.c)
    target_include_directories(voice_opencl_shim PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/OpenCL-Headers)
    target_link_libraries(voice_opencl_shim PRIVATE dl)

    set(OpenCL_INCLUDE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/OpenCL-Headers CACHE PATH "" FORCE)
    set(OpenCL_LIBRARY voice_opencl_shim CACHE STRING "" FORCE)

    set(GGML_OPENCL ON CACHE BOOL "" FORCE)
    # Qualcomm's Adreno-tuned kernel set is OFF: on the S25 Ultra's driver (OpenCL 3.0
    # QUALCOMM build 0800.64.8, compiler E031.47.18.56) compiling gemv_noshuffle_q5_k_f32
    # (kernel 140 of 141) crashes the whole process. The generic OpenCL kernels compiled fine.
    set(GGML_OPENCL_USE_ADRENO_KERNELS OFF CACHE BOOL "" FORCE)
    set(GGML_OPENCL_EMBED_KERNELS ON CACHE BOOL "" FORCE)
    set(TRANSCRIBE_VULKAN OFF CACHE BOOL "" FORCE)
endif()

add_subdirectory(transcribe.cpp)

set(CMAKE_CXX_FLAGS "-O3 -Wall -Wextra -g3")

# Creates and names a library, sets it as either STATIC
# or SHARED, and provides the relative paths to its source code.
# You can define multiple libraries, and CMake builds them for you.
# Gradle automatically packages shared libraries with your APK.
#
# In this top level CMakeLists.txt, ${CMAKE_PROJECT_NAME} is used to define
# the target library name; in the sub-module's CMakeLists.txt, ${PROJECT_NAME}
# is preferred for the same purpose.
#
# In order to load a library into your app from Java/Kotlin, you must call
# System.loadLibrary() and pass the name of the library defined here;
# for GameActivity/NativeActivity derived applications, the same library name must be
# used in the AndroidManifest.xml file.
add_library(${CMAKE_PROJECT_NAME} SHARED
    # List C/C++ source files with relative paths to this CMakeLists.txt.
    voiceinput.cpp
    jni_common.cpp)

if(ANDROID_ABI STREQUAL armeabi-v7a)
    set_target_properties(${CMAKE_PROJECT_NAME} PROPERTIES COMPILE_FLAGS -mfpu=neon)
endif()


target_include_directories(${CMAKE_PROJECT_NAME}
PRIVATE
    transcribe.cpp/include
    transcribe.cpp/ggml/include)

# Specifies libraries CMake should link to your target library. You
# can link libraries from various origins, such as libraries defined in this
# build script, prebuilt third-party libraries, or Android system libraries.
# The keyboard's own native library (libjni_latinime.so) contains an older copy of ggml
# used by the typing model. Hide every symbol of this library's static dependencies and bind
# all internal references locally, so the two ggml versions can never be mixed up at runtime.
target_link_options(${CMAKE_PROJECT_NAME} PRIVATE
    "-Wl,--exclude-libs,ALL"
    "-Wl,-Bsymbolic"
    "-Wl,--gc-sections")
set_target_properties(${CMAKE_PROJECT_NAME} PROPERTIES
    C_VISIBILITY_PRESET hidden
    CXX_VISIBILITY_PRESET hidden)

if(TARGET voice_opencl_shim)
    # For voice_opencl_cache_stats() (GPU program cache diagnostics)
    target_link_libraries(${CMAKE_PROJECT_NAME} voice_opencl_shim)
endif()

if(ANDROID)
    target_link_libraries(${CMAKE_PROJECT_NAME}
        android
        ggml
        transcribe
        log)
else()
    # Desktop build, used only by the custom build's host tests (JVM unit test loads this
    # library and runs the real WhisperGGML class against a real model and audio file)
    find_package(JNI REQUIRED)
    target_include_directories(${CMAKE_PROJECT_NAME} PRIVATE ${JNI_INCLUDE_DIRS})
    target_compile_definitions(${CMAKE_PROJECT_NAME} PRIVATE HOST_TOOL)
    target_link_libraries(${CMAKE_PROJECT_NAME} ggml transcribe)
endif()
